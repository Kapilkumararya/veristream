React Native Mobile Client Integration

        
VeriStream exposes dedicated mobile endpoints for React Native apps to stream investigations, receive real-time push notifications over WebSockets, and render citation cards.

        
        

          React Native Integration Hook (`useVeriStream.ts`):
          
import { useState, useEffect } from 'react';

export const useVeriStream = (endpoint = 'http://localhost:8000') => {
  const [investigation, setInvestigation] = useState(null);
  const [loading, setLoading] = useState(false);

  const investigateClaim = async (text: string) => {
    setLoading(true);
    const response = await fetch(`${endpoint}/api/mobile/investigate`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ text })
    });
    const result = await response.json();
    setInvestigation(result);
    setLoading(false);
  };

  return { investigation, loading, investigateClaim };
};
